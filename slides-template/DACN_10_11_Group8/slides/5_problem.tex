\begin{frame}{Một số khó khăn hiện tại}

\begin{itemize}
    \item \textbf{Thử nghiệm huấn luyện lâu hơn}
    \begin{itemize}
        \item Thử nghiệm với 1.5 và 3 epochs.
        \item Cần GPU có dung lượng VRAM khoảng 24GB trở lên.
    \end{itemize}

    \vspace{0.3cm}

    \item \textbf{Đánh giá chi phí huấn luyện}
    \begin{itemize}
        \item Hiện tại, chi phí được ước lượng dựa trên số lượng
              trajectories và số lượng tham số được huấn luyện.
        \item Nếu có điều kiện, tìm hiểu thêm các phương pháp đo lường
              chi phí huấn luyện chính xác hơn.
    \end{itemize}
\end{itemize}

\end{frame}
