\section{Giới thiệu}

%------------------------------------------------------------
\begin{frame}
    \frametitle{Nội dung báo cáo tuần này}

    \begin{block}{Định hướng}
        Tập trung tìm hiểu sâu hơn các phương pháp hiện có
        và hoàn thiện hướng tiếp cận cho đề tài.
    \end{block}

    \vspace{0.45cm}

    \textbf{Nội dung chính:}

    \begin{enumerate}
        \item \textbf{Phân tích các phương pháp hiện có}

        \vspace{0.25cm}

        Tập trung vào cơ chế retrieval, cách học retrieval
        decision và những hạn chế của các phương pháp hiện tại.

        \vspace{0.3cm}

        \item \textbf{Đề xuất hướng tiếp cận}

        \vspace{0.25cm}

        Đề xuất kiến trúc Adaptive Retrieval Timing và
        cách kiểm chứng các giả thuyết của đề tài.
    \end{enumerate}

\end{frame}
